NOTE: The maintained and recommended launcher is start-win10-kvm.sh.
The commands below are retained as development notes only.

### INSTALL IN TERMUX

pkg update
pkg install qemu-system-aarch64-headless qemu-utils


### WIN ISO DVD

su -c "
/system/bin/env \
LD_PRELOAD=/data/local/tmp/libkvm_flash0_only.so \
/data/data/com.termux/files/usr/bin/qemu-system-aarch64 \
  -name Win10-ARM64 \
  -machine virt,accel=kvm,gic-version=host \
  -cpu host \
  -smp 4,sockets=1,cores=4,threads=1 \
  -m 4096 \
  -drive if=pflash,format=raw,unit=0,readonly=on,file=/data/data/com.termux/files/usr/share/qemu/edk2-aarch64-code.fd \
  -drive if=pflash,format=raw,unit=1,file=/data/media/0/Limbo/QEMU_VARS.fd \
  -device ich9-ahci,id=sata \
  -drive if=none,id=system,format=qcow2,file=/data/media/0/Limbo/win.qcow2,cache=writeback,aio=threads,discard=unmap \
  -device ide-hd,drive=system,bus=sata.0,bootindex=2 \
  -device qemu-xhci,id=xhci \
  -drive if=none,id=installer,format=raw,media=cdrom,readonly=on,file=/data/media/0/Limbo/win10.iso \
  -device usb-storage,drive=installer,bus=xhci.0,bootindex=1,removable=on \
  -device usb-kbd,bus=xhci.0 \
  -device usb-tablet,bus=xhci.0 \
  -device ramfb \
  -netdev user,id=net0 \
  -device e1000,netdev=net0 \
  -device virtio-rng-pci \
  -rtc base=localtime \
  -object secret,id=vncpass,data=1234 \
  -audiodev pa,id=snd0,server=127.0.0.1 \
  -device usb-audio,audiodev=snd0,bus=xhci.0 \
  -display vnc=127.0.0.1:8,password-secret=vncpass \
  -monitor tcp:127.0.0.1:55555,server=on,wait=off
"
